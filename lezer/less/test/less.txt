# Variables

@link-color:        #428bca; // sea blue
@link-color-hover:  darken(@link-color, 10%);

==>

  Declaration(AtKeyword,":",ColorLiteral),";",
  LineComment,
  Declaration(AtKeyword,":",
    CallExpression(
      ValueName,ArgList("(",AtKeyword,",",NumberLiteral(Unit),")"))),";"
)
